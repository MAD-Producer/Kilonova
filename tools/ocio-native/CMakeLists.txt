cmake_minimum_required(VERSION 3.18)
project(MotionOcioBridge LANGUAGES CXX)
find_package(OpenColorIO 2.6.0 EXACT CONFIG REQUIRED NO_DEFAULT_PATH PATHS "${OCIO_PREFIX}/lib/cmake/OpenColorIO")
add_library(motion_ocio_bridge SHARED bridge.cpp)
target_compile_features(motion_ocio_bridge PRIVATE cxx_std_17)
target_link_libraries(motion_ocio_bridge PRIVATE OpenColorIO::OpenColorIO)
if(APPLE)
  set_target_properties(motion_ocio_bridge PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(motion_ocio_bridge PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS motion_ocio_bridge RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
